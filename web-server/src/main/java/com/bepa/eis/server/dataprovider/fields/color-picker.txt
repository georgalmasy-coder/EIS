<script>
const select = document.getElementById('colorName');
const picker = document.getElementById('colorPicker');

// Når brugeren vælger et farvenavn
select.addEventListener('change', () => {
    picker.value = select.value;
});

// Når brugeren justerer farven manuelt
picker.addEventListener('input', () => {
    console.log('Selected color:', picker.value);
});
</script>
